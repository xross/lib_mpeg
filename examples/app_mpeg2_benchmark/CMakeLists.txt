cmake_minimum_required(VERSION 3.21)
include($ENV{XMOS_CMAKE_PATH}/xcommon.cmake)
project(app_mpeg2_benchmark)
set(MPEG_BACKEND libmpeg2 CACHE STRING "Backend required by this MPEG-2 example")
if(NOT MPEG_BACKEND STREQUAL "libmpeg2")
    message(FATAL_ERROR "app_mpeg2_benchmark requires MPEG_BACKEND=libmpeg2")
endif()
set(XMOS_SANDBOX_DIR ${CMAKE_CURRENT_LIST_DIR}/../../..)
set(APP_HW_TARGET XK-EVK-XU316)
set(APP_INCLUDES src)
set(APP_C_SRCS src/entry.c src/mpeg2_benchmark.c src/arena.c)
set(APP_XC_SRCS src/main.xc)
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/optimization.cmake)
set(APP_COMPILER_FLAGS ${MPEG_COLD_OPT} -g -report -fxscope -mcmodel=large -Wno-xcore-fptrgroup)
option(MPEG_BENCH_ALLOW_MPEG1 "Allow identical MPEG-1 input in matched comparisons" OFF)
if(MPEG_BENCH_ALLOW_MPEG1)
    list(APPEND APP_COMPILER_FLAGS -DMPEG_BENCH_ALLOW_MPEG1=1)
endif()
include(${CMAKE_CURRENT_LIST_DIR}/../deps.cmake)
set(APP_XSCOPE_SRCS config.xscope)
XMOS_REGISTER_APP()
set_target_properties(${APP_BUILD_TARGETS} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
