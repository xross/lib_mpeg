cmake_minimum_required(VERSION 3.21)
include($ENV{XMOS_CMAKE_PATH}/xcommon.cmake)
project(test_mpeg_audio)
set(APP_HW_TARGET XK-EVK-XU316 CACHE STRING "Target/XN matching the board's xSCOPE wiring")
option(AUDIO_TEST_XSCOPE "Use hardware xSCOPE output (disable for simulator stdout)" ON)
if(AUDIO_TEST_XSCOPE)
    set(APP_XSCOPE_SRCS ../app_mp2_bench/config.xscope)
endif()
set(APP_C_SRCS ../../third_party/pl_mpeg/tests/audio_s32_test.c)
set(APP_ASM_SRCS ../../third_party/pl_mpeg/xcore/pl_mpeg_audio_xs3.S)
set(APP_INCLUDES ../../third_party/pl_mpeg)
set(flags -O3 -g -report -mcmodel=large -mdual-issue -fwrapv -Wno-xcore-fptrgroup)
set(APP_COMPILER_FLAGS_float ${flags} -DPLM_AUDIO_FORMAT=0)
set(APP_COMPILER_FLAGS_s32_scalar ${flags} -DPLM_AUDIO_FORMAT=1 -DPLM_AUDIO_SYNTHESIS=0)
set(APP_COMPILER_FLAGS_s32_vector ${flags} -DPLM_AUDIO_FORMAT=1 -DPLM_AUDIO_SYNTHESIS=1)
set(APP_COMPILER_FLAGS_s32_vector_smoke ${flags} -DPLM_AUDIO_FORMAT=1 -DPLM_AUDIO_SYNTHESIS=1 -DPLM_AUDIO_TEST_QUICK=1)
XMOS_REGISTER_APP()
foreach(target ${APP_BUILD_TARGETS})
    string(REPLACE "test_mpeg_audio_" "" configuration "${target}")
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/${configuration}")
endforeach()
